# cudart functions whose types the generator would forward, but whose meaning
# the front does not keep across a restart yet.  Each aborts with its name.
# Functions with other handles, handle results or callbacks in their types
# abort without being listed here (front_auto.h).

# They reset or tear down the device, or share it with other processes.
cudaDeviceReset
cudaThreadExit
cudaIpcGetEventHandle
cudaIpcOpenEventHandle

# Memory that the VMM allocator does not make yet.
cudaMallocManaged
cudaMalloc3D

# Their structures hold handles, or they hand out lower-half pointers.
cudaLaunchKernelExC
cudaLaunchKernelExC_ptsz
cudaLaunchCooperativeKernelMultiDevice
cudaGetExportTable

# Texture and surface values must come back exactly (fillers, Phase 2).
cudaCreateTextureObject
cudaDestroyTextureObject
cudaCreateSurfaceObject
cudaDestroySurfaceObject

# Graphs come later; refuse the capture at its start.
cudaStreamBeginCapture
cudaStreamBeginCapture_ptsz
cudaStreamBeginCaptureToGraph
cudaStreamBeginCaptureToGraph_ptsz
